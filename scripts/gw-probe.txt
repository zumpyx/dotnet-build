name: Build .NET Tools

on:
  schedule:
    - cron: '0 2 * * 1'
  workflow_dispatch:
  push:
    branches:
      - main

permissions:
  contents: write

jobs:
  build:
    runs-on: windows-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Build tools
        shell: pwsh
        run: pwsh -NoProfile -File ./scripts/build.ps1
